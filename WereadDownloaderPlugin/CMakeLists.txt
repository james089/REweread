cmake_minimum_required(VERSION 3.16)
project(WereadDownloaderPlugin VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Xml Network)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Xml Network)
find_package(ZLIB REQUIRED)
add_library(weread_downloader_core STATIC src/epub.cpp src/import_registry.cpp)
target_include_directories(weread_downloader_core PUBLIC src)
target_link_libraries(weread_downloader_core PUBLIC Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Xml ZLIB::ZLIB)
add_executable(weread-downloader-worker src/worker.cpp src/native_address.cpp)
target_link_libraries(weread-downloader-worker PRIVATE weread_downloader_core Qt${QT_VERSION_MAJOR}::Network)
option(WRDP_BUILD_NATIVE "Build exact-firmware XOVI UI extension" OFF)
if(WRDP_BUILD_NATIVE)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Gui Qml Quick)
    set(QRCODEGEN_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../downloads/sources/QR-Code-generator-1.8.0" CACHE PATH "Verified QR-Code-generator source")
    if(NOT EXISTS "${QRCODEGEN_SOURCE_DIR}/cpp/qrcodegen.cpp")
        message(FATAL_ERROR "Verified QR-Code-generator source is required")
    endif()
    set(CMAKE_AUTOMOC ON)
    set(CMAKE_AUTORCC ON)
    add_library(weread-downloader MODULE native/main.cpp native/controller.cpp native/controller.h native/cover_store.cpp native/cover_store.h
        native/resources.qrc "${QRCODEGEN_SOURCE_DIR}/cpp/qrcodegen.cpp")
    target_include_directories(weread-downloader PRIVATE "${QRCODEGEN_SOURCE_DIR}/cpp")
    target_link_libraries(weread-downloader PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Gui
        Qt${QT_VERSION_MAJOR}::Qml Qt${QT_VERSION_MAJOR}::Quick Qt${QT_VERSION_MAJOR}::Network ${CMAKE_DL_LIBS})
    set_target_properties(weread-downloader PROPERTIES PREFIX "")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(downloader_address_test tests/address_test.cpp src/native_address.cpp)
    target_include_directories(downloader_address_test PRIVATE src)
    target_link_libraries(downloader_address_test PRIVATE Qt${QT_VERSION_MAJOR}::Core)
    add_executable(downloader_upload_test tests/upload_test.cpp)
    target_include_directories(downloader_upload_test PRIVATE src)
    target_link_libraries(downloader_upload_test PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Network)
    add_executable(downloader_core_test tests/core_test.cpp)
    target_link_libraries(downloader_core_test PRIVATE weread_downloader_core)
    if(WRDP_BUILD_NATIVE)
        find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
        add_executable(downloader_ui_test tests/controller_test.cpp native/controller.cpp native/controller.h native/cover_store.cpp "${QRCODEGEN_SOURCE_DIR}/cpp/qrcodegen.cpp")
        target_include_directories(downloader_ui_test PRIVATE native "${QRCODEGEN_SOURCE_DIR}/cpp")
        target_link_libraries(downloader_ui_test PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Gui Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Qml Qt${QT_VERSION_MAJOR}::Quick Qt${QT_VERSION_MAJOR}::Test)
        add_executable(downloader_cover_test tests/cover_test.cpp native/cover_store.cpp)
        target_include_directories(downloader_cover_test PRIVATE native)
        target_link_libraries(downloader_cover_test PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Gui Qt${QT_VERSION_MAJOR}::Network)
        add_executable(downloader_native_smoke tests/native_smoke.cpp)
        target_link_libraries(downloader_native_smoke PRIVATE Qt${QT_VERSION_MAJOR}::Gui
            Qt${QT_VERSION_MAJOR}::Qml Qt${QT_VERSION_MAJOR}::Quick Qt${QT_VERSION_MAJOR}::Network ${CMAKE_DL_LIBS})
    endif()
    if(NOT CMAKE_CROSSCOMPILING)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME generate_downloader_fixtures COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fixtures.py ${CMAKE_CURRENT_BINARY_DIR}/synthetic-fixtures)
        set_tests_properties(generate_downloader_fixtures PROPERTIES FIXTURES_SETUP downloader_fixtures)
        add_test(NAME downloader_core COMMAND downloader_core_test ${CMAKE_CURRENT_BINARY_DIR}/synthetic-fixtures)
        set_tests_properties(downloader_core PROPERTIES FIXTURES_REQUIRED downloader_fixtures)
    endif()
endif()
