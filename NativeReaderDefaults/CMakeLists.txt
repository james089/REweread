cmake_minimum_required(VERSION 3.16)
project(NativeReaderDefaults LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Qml Quick)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Qml Quick)
add_library(native-reader-defaults MODULE defaults.cpp defaults.h main.cpp)
target_link_libraries(native-reader-defaults PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Qml ${CMAKE_DL_LIBS})
set_target_properties(native-reader-defaults PROPERTIES PREFIX "")
add_executable(native_reader_defaults_test tests.cpp defaults.cpp defaults.h)
target_link_libraries(native_reader_defaults_test PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Qml Qt${QT_VERSION_MAJOR}::Quick)
add_executable(native_reader_defaults_smoke smoke.cpp)
target_link_libraries(native_reader_defaults_smoke PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Qml ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME native_reader_defaults COMMAND native_reader_defaults_test "${CMAKE_CURRENT_SOURCE_DIR}/defaults.qmd")
set_tests_properties(native_reader_defaults PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_test(NAME native_reader_defaults_module COMMAND native_reader_defaults_smoke "$<TARGET_FILE:native-reader-defaults>")
